% --------------------------------------------------------------------------

% One macro draws every MIM cross-section, so a matched footprint is a
% property of the source rather than of four hand-kept coordinate sets.
\newcommand{\MimCell}[2]{%
  \fill[#2] (#1,\mimBotInner) rectangle ({#1+\mimCellWidth},\mimTopInner);
  \fill[electrode] (#1,\mimTopInner) rectangle
    ({#1+\mimCellWidth},\mimTopOuter);
  \fill[electrode] (#1,\mimBotOuter) rectangle
    ({#1+\mimCellWidth},\mimBotInner);
  \draw[structure] (#1,\mimBotOuter) rectangle
    ({#1+\mimCellWidth},\mimTopOuter);
  \draw[interface] (#1,\mimTopInner)--({#1+\mimCellWidth},\mimTopInner)
    (#1,\mimBotInner)--({#1+\mimCellWidth},\mimBotInner);
  \draw[fieldGuide] ({#1+\fieldOffset},1.89)--({#1+\fieldOffset},2.83);
}

% One declaration owns every localized-state position (y / x-offset / length).

    \draw[trapState] ({\cellx+\sx-0.5*\slen},\sy)--({\cellx+\sx+0.5*\slen},\sy);
  }
}
% Occupancy is the only per-state variable: an occupied site keeps the same bar
% and gains a sign-neutral filled marker.  Presence/absence of the marker, not

    ({\stateProgressLeft+\sx+0.17},\sy);
  \fill[cRed!72!black] ({\stateProgressLeft+\sx},\sy) circle [radius=0.068];
}
